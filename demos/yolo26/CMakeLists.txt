cmake_minimum_required(VERSION 3.13)
project(yolo26_streams LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PARSER_ONLY "Build the SDK-independent compact-output test only" OFF)
include(CTest)
if(BUILD_TESTING OR PARSER_ONLY)
  add_executable(yolo26_compact_test test_compact.cpp)
  if(MSVC)
    target_compile_options(yolo26_compact_test PRIVATE /W4 /UNDEBUG)
  else()
    target_compile_options(yolo26_compact_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  enable_testing()
  add_test(NAME yolo26_compact_output COMMAND yolo26_compact_test)
endif()
if(PARSER_ONLY)
  return()
endif()

set(SOPHON_DEMO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/sophon-demo" CACHE PATH "Official SOPHON-DEMO checkout")
# Reuse the same SDK wrappers as the other PCIe demos in this repository.
# No detector source from another demo is required.
set(YOLO_HEADER_DIR "${SOPHON_DEMO_DIR}/sample/YOLOv8_plus_det/cpp/yolov8_bmcv" CACHE PATH "Directory containing bm_wrapper.hpp, utils.hpp and json.hpp")
foreach(header bm_wrapper.hpp utils.hpp json.hpp)
  if(NOT EXISTS "${YOLO_HEADER_DIR}/${header}")
    message(FATAL_ERROR "Missing ${header}; prepare SOPHON-DEMO or set YOLO_HEADER_DIR to its utility header directory")
  endif()
endforeach()
set(FFMPEG_DIR /opt/sophon/sophon-ffmpeg-latest/lib/cmake CACHE PATH "Sophon FFmpeg config directory")
set(OpenCV_DIR /opt/sophon/sophon-opencv-latest/lib/cmake/opencv4 CACHE PATH "Sophon OpenCV config directory")
find_package(FFMPEG REQUIRED)
find_package(OpenCV REQUIRED)
find_package(libsophon REQUIRED)
find_package(Threads REQUIRED)
link_directories(${FFMPEG_LIB_DIRS} ${OpenCV_LIB_DIRS} ${LIBSOPHON_LIB_DIRS})
add_executable(yolo26_streams.pcie main.cpp detector/yolo26_det.cpp)
target_include_directories(yolo26_streams.pcie PRIVATE detector "${YOLO_HEADER_DIR}" ${FFMPEG_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${LIBSOPHON_INCLUDE_DIRS})
target_compile_options(yolo26_streams.pcie PRIVATE -O2 -UNDEBUG -Wall -Wextra)
target_link_libraries(yolo26_streams.pcie ${FFMPEG_LIBS} ${OpenCV_LIBS} ${the_libbmlib.so} ${the_libbmrt.so} ${the_libbmcv.so} Threads::Threads)
